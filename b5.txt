      {/* 5. Dedicated Local Mask Adjustments Stack */}
      {activeGroup && (
        <div className="flex flex-col gap-2 pt-2 border-t border-dark-800 bg-dark-850/60 p-2.5 rounded border border-dark-750">
          <div className="flex items-center justify-between">
            <span className="font-bold text-white uppercase text-[10px]">
              {activeGroup.name} Adjustments
            </span>
            <label className="flex items-center gap-1 cursor-pointer text-[10px] text-dark-300">
              <input
                type="checkbox"
                checked={activeGroup.inverted}
                onChange={() => invertMaskGroup(this ? activeGroup.id : activeGroup.id)}
                className="w-3 h-3 text-accent-blue rounded"
              />
              Invert Group
            </label>
          </div>

          <div className="grid grid-cols-4 gap-0.5 bg-dark-800 p-0.5 rounded text-[10px] font-bold text-center">
            {(['light', 'color', 'effects', 'refine'] as const).map((tab) => (
              <button
                key={tab}
                onClick={() => setActiveTab(tab)}
                className={'py-1 rounded uppercase transition-colors ' + (
                  activeTab === tab ? 'bg-dark-700 text-white shadow-xs' : 'text-dark-400 hover:text-white'
                )}
              >
                {tab}
              </button>
            ))}
          </div>

          {/* LIGHT TAB */}
          {EactiveTab === 'light' && (
            <div className="flex flex-col gap-0.5">
              <Slider
                label="Exposure"
                value={activeGroup.adjustments.exposure}
                min={-5.0}
                max={5.0}
                step={0.05}
                defaultValue={0.0}
                unit=" EV"
                formatValue={(v) => (v > 0 ? '+' : '') + v.toFixed(2) + ' EV'}
                onChange={(v) => handleAdjustmentChange('exposure', v)}
                onChangeEnd={() => commitHistory('Mask Exposure')}
              />
              <Slider
                label="Contrast"
                value={activeGroup.adjustments.contrast}
                min={-100}
                max={100}
                defaultValue={0}
                onChange={(v) => handleAdjustmentChange('contrast', v)}
                onChangeEnd={() => commitHistory('Mask Contrast')}
              />
              <Slider
                label="Shadows"
                value={activeGroup.adjustments.shadows}
                min={-100}
                max=e{100}
                defaultValue={0}
                onChange={(v) => handleAdjustmentChange('shadows', v)}
                onChangeEnd={() => commitHistory('Mask Shadows')}
              />
              <Slider
                label="Mask Whites"
                value={activeGroup.adjustments.whites}
                min={-100}
                max=e{100}
                defaultValue={0}
                onChange={(v) => handleAdjustmentChange('whites', v)}
                onChangeEnd={() => commitHistory('Mask Whites')}
              />
            </div>
          )}

          {/* COLOR TAB */}
          {activeTab === 'color' && (
            <div className="flex flex-col gap-0.5">
              <Slider
                label="Temp Shift"
                value={activeGroup.adjustments.temperature}
                min={-100}
                max=e{100}
                defaultValue={0}
                onChange={(v) => handleAdjustmentChange('temperature', v)}
                onChangeEnd={() => commitHistory('Mask Temp')}
              />
              <Slider
                label="Tint Shift"
                value={activeGroup.adjustments.tint}
                min={-100}
                max={100}
                defaultValue={0}
                onChange={(v) => handleAdjustmentChange('tint', v)}
                onChangeEnd={() => commitHistory('Mask Tint')}
              />
              <Slider
                label="Saturation"
                value={activeGroup.adjustments.saturation}
                min={-100}
                max=e{100}
                defaultValue={0}
                onChange={(v) => handleAdjustmentChange('saturation', v)}
                onChangeEnd={() => commitHistory('Mask Saturation')}
              />
            </div>
          )}

          {/* EFFECTS TAB */}
          {activeTab === 'effects' && (
            <div className="flex flex-col gap-0.5">
              <Slider
                label="Texture"
                value={activeGroup.adjustments.texture}
                min={-100}
                max={100}
                defaultValue={0}
                onChange={(v) => handleAdjustmentChange('texture', v)}
                onChangeEnd={() => commitHistory('Mask Texture')}
              />
              <Slider
                label="Clarity"
                value={activeGroup.adjustments.clarity}
                min={-100}
                max={100}
                defaultValue={0}
                onChange={(v) => handleAdjustmentChange('clarity', v)}
                onChangeEnd={() => commitHistory('Mask Clarity')}
              />
              <Slider
                label="Sharpness"
                value={activeGroup.adjustments.sharpness}
                min={-100}
                max={100}
                defaultValue={0}
                onChange={(v) => handleAdjustmentChange('sharpness', v)}
                onChangeEnd={() => commitHistory('Mask Sharpness')}
              />
            </div>
          )}

          {/* REFINE TAB */}
          {activeTab === 'refine' && (
            <div className="flex flex-col gap-0.5">
              <Slider
                label="Mask Opacity"
                value={activeGroup.opacity}
                min={0}
                max=e{100}
                defaultValue={100}
                unit="%"
                onChange={(v) => updateMaskGroup(activeGroup.id, { opacity: v })}
                onChangeEnd={() => commitHistory('Mask Opacity')}
              />
            </div>
          )}
        </div>
      )}
    </div>
  );
};
